using StateMachine;
using StateMachineTest.TestInfrastructure;

namespace StateMachineTest;

public class ModuleScriptFacts
{
    private const string CheckerModule = """
        module Checker {
            input Entry;
            output Done, Failed;
            terminal Exit;

            def Entry(Step) { 1->Next; 3->Failed; }
            def Exit(Step) { 1->Done; }
            Next->Entry to Exit;
        }
        """;

    [Fact]
    public void TransitionFromModuleInstance_RejectsEventNotPublishedByTerminal()
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        Action create = () => engine.CreateStateMachine("""
            import Checker;
            def Check(Checker) { Done->Accepted; }
            def Target(Step) { }
            WrongEvent->Check to Target;
            """);

        // 不能为了连线成功而把唯一 terminal 的 Accepted 分支改成 WrongEvent。
        create.Should().Throw<ScriptException>()
            .WithMessage("*Check*terminal*WrongEvent*");
    }

    [Theory]
    [InlineData("Check.Next")]
    [InlineData("check.next")]
    public void OutputMapping_RejectsCollisionWithInternalEvent(string externalEvent)
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        Action create = () => engine.CreateStateMachine($$"""
            import Checker;
            def Check(Checker) { Done->{{externalEvent}}; }
            """);

        create.Should().Throw<ScriptException>()
            .WithMessage($"*Check*output*{externalEvent}*碰撞*");
    }

    [Theory]
    [InlineData("", -1)]
    [InlineData("", 2)]
    [InlineData("branch ", -1)]
    [InlineData("branch ", 2)]
    public void OutputMapping_RejectsIndicesOutsideDeclaredOutputs(string branchKeyword, int index)
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        Action create = () => engine.CreateStateMachine($$"""
            import Checker;
            def Check(Checker) { {{branchKeyword}}{{index}}->Accepted; }
            """);

        // Checker 只有两个出口，合法索引为 0 和 1。
        create.Should().Throw<ScriptException>()
            .WithMessage($"*Check*output*{index}*");
    }

    [Fact]
    public void OutputMapping_RejectsUndeclaredOutputName()
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        Action create = () => engine.CreateStateMachine("""
            import Checker;
            def Check(Checker) { Missing->Accepted; }
            """);

        create.Should().Throw<ScriptException>()
            .WithMessage("*Check*output*Missing*");
    }

    [Fact]
    public void TransitionFromModuleInstance_ConnectsActualOutputPublisherNotUnrelatedTerminal()
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        engine.CreateStateMachine("""
            import Checker;
            def Check(Checker) { Done->Accepted; Failed->Rejected; }
            def Success(Step) { }
            def Failure(Step) { }
            Accepted->Check to Success;
            Rejected->Check to Failure;
            """);

        var entry = engine.GetNode("Check.Entry");
        var exit = engine.GetNode("Check.Exit");
        entry.TargetState(engine.GetEvent("Rejected"))
            .Should().BeSameAs(engine.GetNode("Failure"));
        exit.TargetState(engine.GetEvent("Accepted"))
            .Should().BeSameAs(engine.GetNode("Success"));
        exit.HasTransition(engine.GetEvent("Rejected")).Should().BeFalse();
        exit.UpdateEventDescriptions();
        exit.EventDescriptions.Select(d => (d.Index, d.Description)).Should()
            .BeEquivalentTo(new[] { (1, "Accepted") });
    }

    [Fact]
    public void TransitionToModuleInstance_ConnectsToDeclaredInput()
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        engine.CreateStateMachine("""
            import Checker;
            def Start(Step) { 1->Begin; }
            def Check(Checker) { }
            Begin->Start to Check;
            """);

        engine.GetNode("Start").TargetState(engine.GetEvent("Begin"))
            .Should().BeSameAs(engine.GetNode("Check.Entry"));
    }

    [Theory]
    [InlineData("Failed->Rejected; Done->Accepted;")]
    [InlineData("0->Accepted; 1->Rejected;")]
    [InlineData("branch 0->Accepted; branch 1->Rejected;")]
    public void OutputMapping_RebindsBranchesWithoutChangingOtherInstances(string outputMap)
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        engine.CreateStateMachine($$"""
            import Checker;
            def Left(Checker) { {{outputMap}} }
            def Right(Checker) { }
            """);

        var entry = engine.GetNode("Left.Entry");
        var exit = engine.GetNode("Left.Exit");
        var otherExit = engine.GetNode("Right.Exit");
        entry.UpdateEventDescriptions();
        exit.UpdateEventDescriptions();
        otherExit.UpdateEventDescriptions();

        entry.EventDescriptions.Select(d => (d.Index, d.Description)).Should()
            .BeEquivalentTo(new[] { (1, "Left.Next"), (3, "Rejected") });
        exit.EventDescriptions.Select(d => (d.Index, d.Description)).Should()
            .BeEquivalentTo(new[] { (1, "Accepted") });
        otherExit.EventDescriptions.Select(d => (d.Index, d.Description)).Should()
            .BeEquivalentTo(new[] { (1, "Right.Done") });
        engine.GetEventNames().Should().BeEquivalentTo(
            "Left.Next", "Accepted", "Rejected", "Right.Next", "Right.Done", "Right.Failed");
    }

    [Fact]
    public void ModuleInstances_KeepInternalTransitionsWithinEachInstance()
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        engine.CreateStateMachine("""
            import Checker;
            def Left(Checker) { }
            def Right(Checker) { }
            """);

        engine.GetNode("Left.Entry").TargetState(engine.GetEvent("Left.Next"))
            .Should().BeSameAs(engine.GetNode("Left.Exit"));
        engine.GetNode("Right.Entry").TargetState(engine.GetEvent("Right.Next"))
            .Should().BeSameAs(engine.GetNode("Right.Exit"));
        engine.GetNode("Left.Entry").HasTransition(engine.GetEvent("Right.Next"))
            .Should().BeFalse();
        engine.GetNode("Right.Entry").HasTransition(engine.GetEvent("Left.Next"))
            .Should().BeFalse();
    }

    [Fact]
    public void ModuleInstances_KeepNodesAndUnmappedEventsInSeparateNamespaces()
    {
        var engine = new FSMEngine(new ScriptNodeFactory());
        engine.RegisterModule("Checker", CheckerModule);

        engine.CreateStateMachine("""
            import Checker;
            def Left(Checker) { }
            def Right(Checker) { }
            """);

        // 模块模板和视觉代理均不是运行时节点；未映射的出口也应留在实例内。
        engine.GetNodeNames().Should().BeEquivalentTo(
            "Left.Entry", "Left.Exit", "Right.Entry", "Right.Exit");
        engine.GetNode("Left.Entry").Should().NotBeSameAs(engine.GetNode("Right.Entry"));
        engine.GetEventNames().Should().BeEquivalentTo(
            "Left.Next", "Left.Done", "Left.Failed",
            "Right.Next", "Right.Done", "Right.Failed");
        engine.GetEvent("Left.Next").Should().NotBeSameAs(engine.GetEvent("Right.Next"));
    }
}
